\documentclass{article}
\usepackage[margin=5mm]{geometry}
\usepackage{graphicx}

\pagestyle{empty}
\setlength{\parindent}{0pt}
\setlength{\parskip}{0pt}

\newcommand{\includeSortedImage}[1]{%
  \noindent
  \makebox[\linewidth][c]{\includegraphics[width=\linewidth,height=0.45\textheight,keepaspectratio]{images/#1}}%
  \par
  \vspace{5pt}
}

\newcommand{\includeNumberedImage}[1]{%
  \ifnum#1<10
    \edef\imageName{00#1}%
  \else
    \ifnum#1<100
      \edef\imageName{0#1}%
    \else
      \edef\imageName{#1}%
    \fi
  \fi
  \IfFileExists{images/\imageName.jpeg}{\includeSortedImage{\imageName.jpeg}}{}%
  \IfFileExists{images/\imageName.jpg}{\includeSortedImage{\imageName.jpg}}{}%
  \IfFileExists{images/\imageName.pdf}{\includeSortedImage{\imageName.pdf}}{}%
  \IfFileExists{images/\imageName.png}{\includeSortedImage{\imageName.png}}{}%
}

\begin{document}
\newcount\imageIndex
\imageIndex=1
\loop
  \includeNumberedImage{\the\imageIndex}
  \advance\imageIndex by 1
\ifnum\imageIndex<1000
\repeat
\end{document}
