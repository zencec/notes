\newcommand{\includeNumberedImage}[1]{%
  \noindent
  \ifnum#1<10
    \includegraphics[width=\linewidth,height=0.45\textheight,keepaspectratio]{00#1.png}%
  \else
    \ifnum#1<100
      \includegraphics[width=\linewidth,height=0.45\textheight,keepaspectratio]{0#1.png}%
    \else
      \includegraphics[width=\linewidth,height=0.45\textheight,keepaspectratio]{#1.png}%
    \fi
  \fi
  \par\nointerlineskip
}

\begin{document}
\foreach \n in {1,...,65}{\includeNumberedImage{\n}}
\end{document}
