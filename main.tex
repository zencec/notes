\documentclass{article}
\usepackage[margin=5mm]{geometry}
\usepackage{graphicx}

\pagestyle{empty}
\setlength{\parindent}{0pt}
\setlength{\parskip}{0pt}

\newcommand{\includeSortedImage}[1]{%
  \noindent
  \makebox[\linewidth][c]{\includegraphics[width=\linewidth,height=0.45\textheight,keepaspectratio]{images/#1}}%
  \par
  \vspace{5pt}
}

\begin{document}
\directlua{
  local lfs = require("lfs")
  local images = {}

  for filename in lfs.dir("images") do
    local lastDot = 0
    for position = 1, string.len(filename) do
      if string.byte(filename, position) == 46 then
        lastDot = position
      end
    end
    local extension = string.lower(string.sub(filename, lastDot + 1))
    if extension == "png" or extension == "jpg" or extension == "jpeg" or extension == "pdf" then
      table.insert(images, filename)
    end
  end

  table.sort(images)
  for _, filename in ipairs(images) do
    tex.sprint(string.char(92) .. "includeSortedImage{")
    tex.sprint(-2, filename)
    tex.sprint("}")
  end
}
\end{document}
